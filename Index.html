Answer questions clearly and directly. ${needsVideos && videos.length > 0 ? `The user requested video recommendations. The following YouTube videos were retrieved: ${JSON.stringify(videos.map(v => v.title))}. Summarize an answer and briefly reference the topics covered in these videos.` : 'If the user does not request video materials, simply answer the question thoroughly without mentioning video searches.'}`; const geminiResponse = await ai.models.generateContent({ model: 'gemini-2.5-flash', contents: [ { role: 'user', parts: [{ text: `${systemPrompt}\n\nUser Question: ${prompt}` }] } ] }); res.json({ reply: geminiResponse.text, videos: videos }); } catch (error) { console.error('Gemini error:', error); res.status(500).json({ error: 'Failed to process request.' }); } }); // Single-page UI delivery app.get('/', (req, res) => { res.type('html').send(` AI Chatbox

`); }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Server live at http://localhost:${PORT}`); });
